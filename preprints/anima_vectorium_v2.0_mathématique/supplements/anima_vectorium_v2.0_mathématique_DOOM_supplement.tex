\chapter{Supplement: Formalization of DOOM as a Geometric–Operator System}

\section{Overview}

This supplement provides a rigorous mathematical encoding of the classical
first-person game DOOM within the Anima-Vectorium v2.0 Mathématique framework.
The purpose is to demonstrate that interactive environments can be represented
as expressive and geometric manifolds equipped with operator algebra, kernel
flow, and homotopy-invariant safety constraints.

\section{DOOM State Manifold}

\begin{definition}[DOOM State Manifold]
Let $\mathcal{D}$ denote the DOOM state space. We define the DOOM manifold
\[
D_M = (M_D, g_D)
\]
where $M_D$ is the configuration space of player position, orientation,
health, inventory, and local environment state, and $g_D$ is a Riemannian
metric encoding spatial adjacency and movement cost.
\end{definition}

The manifold $M_D$ is piecewise-smooth due to discrete transitions such as
door opening and enemy spawning. We treat these transitions as smooth charts
on overlapping regions.

\section{Expressive Encoding of DOOM Inputs}

\begin{definition}[Expressive Input Manifold]
Let $E_D$ be the expressive manifold of DOOM inputs:
\[
E_D = \{ \texttt{MoveForward}, \texttt{TurnLeft}, \texttt{Fire}, 
\texttt{OpenDoor}, \texttt{Use} \}.
\]
Each input is represented as a smooth expressive vector in $T(E_D)$.
\end{definition}

\section{Projection Functor for DOOM}

\begin{definition}[DOOM Projection Functor]
Define
\[
\Pi_D : \mathbf{Expr}_D \to \mathbf{Geom}_D
\]
mapping expressive input vectors to geometric state transitions in $D_M$.
\end{definition}

\begin{theorem}[Naturality of DOOM Projection]
For any expressive DOOM input $f : E_D \to E_D$,
\[
\Pi_D(f) \circ \eta_{E_D} = \eta_{E_D} \circ f.
\]
\end{theorem}

\section{DOOM Operator Algebra}

We define the following operators acting on $T(D_M)$:

\[
\mathcal{O}_D = 
\{\mathrm{Move}, \mathrm{Turn}, \mathrm{Fire}, \mathrm{Open}, \mathrm{AI}\}.
\]

\begin{definition}[DOOM Operator Lie Algebra]
The Lie algebra $\mathfrak{A}_D$ is generated by $\mathcal{O}_D$ with bracket
\[
[X,Y] = XY - YX.
\]
\end{definition}

\begin{theorem}[Non-Commutation of Movement and Firing]
\[
[\mathrm{Move}, \mathrm{Fire}] \neq 0.
\]
\end{theorem}

\begin{theorem}[Commutation of Door Opening and Firing]
\[
[\mathrm{Open}, \mathrm{Fire}] = 0.
\]
\end{theorem}

\section{DOOM Kernel Flow}

\begin{definition}[DOOM Kernel]
The DOOM kernel flow $K_D : D_M \to D_M$ is generated by the vector field
$X_D$ encoding deterministic game physics:
\[
\frac{d}{dt} \phi_t(s) = X_D(\phi_t(s)).
\]
\end{definition}

\begin{theorem}[Deterministic DOOM Evolution]
If $\phi_t(s_1) = \phi_t(s_2)$ for all $t$, then $s_1 = s_2$.
\end{theorem}

\section{Safety Region in DOOM}

\begin{definition}[DOOM Safety Region]
Define
\[
\mathrm{SafeRegion}_D = \{ s \in D_M : \mathrm{Health}(s) > 0 \}.
\]
\end{definition}

\begin{definition}[Safety Homotopy for DOOM]
A homotopy $H_D : D_M \times [0,1] \to D_M$ is safety-preserving if
\[
H_D(s,t) \in \mathrm{SafeRegion}_D.
\]
\end{definition}

\begin{theorem}[Safety Invariance Under Kernel Flow]
If $s \in \mathrm{SafeRegion}_D$, then
\[
\phi_t(s) \in \mathrm{SafeRegion}_D \quad \forall t.
\]
\end{theorem}

\section{Category-Theoretic Structure of DOOM}

\begin{theorem}[DOOM Adjunction]
The expressive-to-geometric alignment in DOOM satisfies
\[
\Pi_D \dashv K_D.
\]
\end{theorem}

\begin{center}
\begin{tikzcd}
E_D \arrow[r,"f"] \arrow[d,"\eta_{E_D}"'] & E_D' \arrow[d,"\eta_{E_D'}"] \\
\Pi_D(E_D) \arrow[r,"\Pi_D(f)"'] & \Pi_D(E_D')
\end{tikzcd}
\end{center}

\section{Verification via DOOM Bisimulation}

\begin{definition}[DOOM Bisimulation]
Define relation $R_D$ by
\[
(s_1, s_2) \in R_D \iff K_D(s_1) = K_D(s_2).
\]
\end{definition}

\begin{theorem}
$R_D$ is a bisimulation on $D_M$.
\end{theorem}

\chapter*{Conclusion}

This supplement demonstrates that DOOM can be rigorously encoded as a
manifold–operator–kernel system within the Anima-Vectorium framework. The
formalization shows that interactive environments can be treated as geometric
objects with expressive inputs, deterministic flows, operator algebra, and
homotopy-invariant safety constraints.
